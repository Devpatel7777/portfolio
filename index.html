WITH ltv AS (
  SELECT c.customer_id, c.city, SUM(oi.net_total) AS total_spent
  FROM customers c
  JOIN orders o ON c.customer_id = o.customer_id
  JOIN order_items oi ON o.order_id = oi.order_id
  GROUP BY 1, 2
)
SELECT city, COUNT(*) AS buyers, ROUND(AVG(total_spent), 0) AS avg_ltv
FROM ltv GROUP BY 1 ORDER BY avg_ltv DESC;
